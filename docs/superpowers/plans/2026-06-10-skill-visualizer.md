# Skill Visualizer Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a single-page static website in `assets/` that visualizes all 1596+ SKILL.md/AGENTS.md files with a tree sidebar, sticky frontmatter header, rendered markdown body, and inline file previews.

**Architecture:** A Node.js build script (`build.js`) scans the repo, parses frontmatter, reads supporting files, and outputs `assets/data.json`. A single `assets/index.html` loads that JSON and renders everything client-side with marked.js + highlight.js via CDN. Deployed on GitHub Pages from `assets/`.

**Tech Stack:** Node.js (gray-matter for YAML parsing), vanilla HTML/CSS/JS, marked.js CDN (markdown), highlight.js CDN (syntax highlighting)

### Task 2: build.js — Core scanner with frontmatter parsing

**Files:**

- [ ] **Step 1: Write build.js with recursive walk and frontmatter parsing**

```js
import { readFileSync, writeFileSync, readdirSync, statSync } from 'fs';
import { join, basename, extname, relative } from 'path';
import matter from 'gray-matter';

const ROOT = '.';
const TARGET_FILES = ['SKILL.md', 'AGENTS.md', 'CLAUDE.md'];
const SKIP_DIRS = new Set(['.git', 'node_modules', '.claude', '.DS_Store']);
const SUPPORTING_DIRS = ['references', 'scripts', 'examples', 'agents', 'assets'];
const CATEGORY_DIRS = [
  'guidelines-mindsets', 'design-creative', 'game-development',
  'ai-tools', 'dev-engineering', 'document-generation',
  'platform-ecosystems', 'life-automation', 'domain-expertise'
];
const COLLECTION_DIRS = [
  'ECC', 'awesome-copilot', 'gstack', 'get-shit-done', 'servers', 'spec-kit'
];

function isTextFile(ext) {
  return ['.md', '.yaml', '.yml', '.json', '.py', '.js', '.ts', '.jsx', '.tsx',
    '.css', '.html', '.txt', '.sh', '.toml', '.cfg', '.ini'].includes(ext);
}

function readFilesRecursive(dirPath, depth = 0) {
  if (depth > 5) return []; // safety limit
  const entries = [];
  try {
    const items = readdirSync(dirPath);
    for (const item of items) {
      if (SKIP_DIRS.has(item) || item.startsWith('.')) continue;
      const fullPath = join(dirPath, item);
      let stat;
      try { stat = statSync(fullPath); } catch { continue; }
      if (stat.isDirectory()) {
        const children = readFilesRecursive(fullPath, depth + 1);
        if (children.length > 0) {
          entries.push({ name: item + '/', type: 'dir', children });
        }
      } else if (stat.isFile()) {
        const ext = extname(item);
        if (isTextFile(ext)) {
          try {
            const content = readFileSync(fullPath, 'utf-8');
            entries.push({ name: item, type: 'file', path: fullPath, ext, content });
          } catch { /* skip unreadable */ }
        }
      }
    }
  } catch { /* skip inaccessible */ }
  return entries;
}

function scanDir(dirPath, type) {
  const node = {
    name: basename(dirPath),
    type,
    path: dirPath,
    children: []
  };

  let items;
  try { items = readdirSync(dirPath); } catch { return node; }

  for (const item of items) {
    if (SKIP_DIRS.has(item) || item.startsWith('.')) continue;
    const fullPath = join(dirPath, item);
    let stat;
    try { stat = statSync(fullPath); } catch { continue; }
    if (!stat.isDirectory()) continue;

    // Look for target files (SKILL.md, AGENTS.md, CLAUDE.md) inside this dir
    const child = { name: item, path: fullPath, type: 'unknown', frontmatter: {}, body: '', files: [] };
    let found = false;

    for (const tf of TARGET_FILES) {
      const tfPath = join(fullPath, tf);
      try {
        const raw = readFileSync(tfPath, 'utf-8');
        const parsed = matter(raw);
        child.type = tf === 'SKILL.md' ? 'skill' : 'agent-doc';
        child.frontmatter = parsed.data;
        child.body = parsed.content;
        child.sourceFile = tf;
        found = true;
        break;
      } catch {}
    }

    if (found) {
      // Enumerate supporting files
      child.files = [];
      for (const sd of SUPPORTING_DIRS) {
        const sdPath = join(fullPath, sd);
        try {
          const sdStat = statSync(sdPath);
          if (sdStat.isDirectory()) {
            const contents = readFilesRecursive(sdPath);
            if (contents.length > 0) {
              child.files.push({ name: sd + '/', type: 'dir', children: contents });
            }
          }
        } catch {}
      }
      node.children.push(child);
    } else {
      // Recurse into subdirectories (for collections like ECC that have nested structure)
      const sub = scanDir(fullPath, 'unknown');
      if (sub.children.length > 0) {
        node.children.push(sub);
      }
    }
  }

  return node;
}

console.log('Scanning repository...');

const tree = [];

// Scan categories
for (const cat of CATEGORY_DIRS) {
  try { statSync(cat); } catch { continue; }
  const node = scanDir(cat, 'category');
  if (node.children.length > 0) tree.push(node);
  console.log(`  ${cat}: ${node.children.length} entries`);
}

// Scan collections
for (const col of COLLECTION_DIRS) {
  try { statSync(col); } catch { continue; }
  const node = scanDir(col, 'collection');
  if (node.children.length > 0) tree.push(node);
  console.log(`  ${col}: ${node.children.length} entries`);
}

const total = tree.reduce((sum, n) => sum + n.children.length, 0);
console.log(`Total: ${total} entries`);

// Compact JSON
const json = JSON.stringify({ tree });
console.log(`Output size: ${(json.length / 1024 / 1024).toFixed(2)} MB`);

writeFileSync('assets/data.json', json);
console.log('Written to assets/data.json');
```

// ==================== DATA & STATE ====================

// ==================== MARKDOWN SETUP ====================
marked.setOptions({
  highlight: function(code, lang) {
    if (lang && hljs.getLanguage(lang)) {
      return hljs.highlight(code, { language: lang }).value;
    }
    return hljs.highlightAuto(code).value;
  },
  breaks: true
});

// ==================== DATA LOADING ====================
async function loadData() {
  try {
    const res = await fetch('data.json');
    data = await res.json();
    renderTree(data.tree);
  } catch (err) {
    sidebarEl.innerHTML = '<div style="padding:16px;color:#ef4444">Failed to load data.json</div>';
  }
}

// ==================== TREE RENDERING ====================
function renderTree(tree, filterText = '') {
  sidebarEl.innerHTML = '';
  const ft = filterText.toLowerCase().trim();

  for (const node of tree) {
    const isCategory = node.type === 'category' || node.type === 'collection';

    // Category/Collection header
    const catDiv = document.createElement('div');
    catDiv.className = 'tree-item tree-category';
    catDiv.innerHTML = `<span class="arrow">▼</span> ${isCategory ? '📁' : '📦'} ${node.name}`;

    const childrenDiv = document.createElement('div');
    childrenDiv.className = 'category-children';

    let visibleCount = 0;
    for (const child of (node.children || [])) {
      const childName = (child.frontmatter?.name || child.name || '');
      const childDesc = (child.frontmatter?.description || '');
      const matches = !ft || childName.toLowerCase().includes(ft) || childDesc.toLowerCase().includes(ft);

      const skillDiv = document.createElement('div');
      skillDiv.className = 'tree-item tree-skill';
      skillDiv.textContent = childName || child.name;
      skillDiv.addEventListener('click', () => selectSkill(child, node.name));
      skillDiv.dataset.path = child.path;

      if (matches) {
        visibleCount++;
      } else {
        skillDiv.style.display = 'none';
      }
      childrenDiv.appendChild(skillDiv);
    }

    catDiv.addEventListener('click', () => {
      catDiv.classList.toggle('collapsed');
      childrenDiv.classList.toggle('collapsed');
    });

    sidebarEl.appendChild(catDiv);
    sidebarEl.appendChild(childrenDiv);

    if (ft) {
      // Auto-expand categories with visible children
      if (visibleCount === 0) {
        catDiv.classList.add('collapsed');
        childrenDiv.classList.add('collapsed');
      }
    }
  }
}

// ==================== SKILL SELECTION ====================
function selectSkill(node, parentName) {
  currentPath = node.path;
  contentEl.innerHTML = '';

  // Highlight active tree item
  document.querySelectorAll('.tree-skill.active').forEach(el => el.classList.remove('active'));
  const activeEl = document.querySelector(`[data-path="${node.path}"]`);
  if (activeEl) activeEl.classList.add('active');

  // Breadcrumb
  breadcrumbEl.innerHTML = `<span onclick="showEmpty()">🏠</span><span class="sep">/</span><span>${parentName}</span><span class="sep">/</span><span>${node.frontmatter?.name || node.name}</span>`;

  // Frontmatter card
  const fm = node.frontmatter || {};
  const fmCard = document.createElement('div');
  fmCard.id = 'fm-card';
  let fmHTML = `<div class="fm-name">${escapeHtml(fm.name || node.name)}</div>`;
  if (fm.description) {
    fmHTML += `<div class="fm-desc">${escapeHtml(fm.description)}</div>`;
  }
  // Other frontmatter keys (beyond name/description)
  const extraKeys = Object.keys(fm).filter(k => k !== 'name' && k !== 'description' && k !== '');
  if (extraKeys.length > 0) {
    fmHTML += '<div class="fm-meta">';
    for (const k of extraKeys) {
      const val = typeof fm[k] === 'string' ? fm[k] : JSON.stringify(fm[k]);
      fmHTML += `<span><strong>${escapeHtml(k)}:</strong> ${escapeHtml(val)}</span>`;
    }
    fmHTML += '</div>';
  }
  fmHTML += `<div class="fm-meta"><span class="fm-badge">${node.type}</span><span>${node.path}</span></div>`;
  fmCard.innerHTML = fmHTML;
  contentEl.appendChild(fmCard);

  // Markdown body
  if (node.body && node.body.trim()) {
    const mdDiv = document.createElement('div');
    mdDiv.id = 'md-body';
    mdDiv.innerHTML = marked.parse(node.body);
    contentEl.appendChild(mdDiv);
  }

  // Files section
  if (node.files && node.files.length > 0) {
    const filesDiv = document.createElement('div');
    filesDiv.id = 'files-section';
    filesDiv.innerHTML = '<h3>📎 Attached Files</h3>';
    renderFileTree(node.files, filesDiv, 0);
    contentEl.appendChild(filesDiv);
  }

  // Scroll to top
  contentEl.scrollTop = 0;
}

function renderFileTree(files, parentEl, depth) {
  for (const f of files) {
    if (f.type === 'dir') {
      const dirLabel = document.createElement('div');
      dirLabel.className = 'file-dir';
      dirLabel.style.paddingLeft = (12 + depth * 16) + 'px';
      dirLabel.textContent = '📁 ' + f.name;
      parentEl.appendChild(dirLabel);
      if (f.children) renderFileTree(f.children, parentEl, depth + 1);
    } else {
      const link = document.createElement('div');
      link.className = 'file-link';
      link.style.paddingLeft = (12 + depth * 16) + 'px';
      const icon = getFileIcon(f.name);
      link.innerHTML = `<span class="icon">${icon}</span> ${f.name}`;
      link.addEventListener('click', () => openFileModal(f));
      parentEl.appendChild(link);
    }
  }
}

function getFileIcon(name) {
  if (name.endsWith('.md')) return '📝';
  if (name.endsWith('.yaml') || name.endsWith('.yml')) return '⚙️';
  if (name.endsWith('.py')) return '🐍';
  if (name.endsWith('.js') || name.endsWith('.ts')) return '📜';
  if (name.endsWith('.css')) return '🎨';
  if (name.endsWith('.html')) return '🌐';
  if (name.endsWith('.json')) return '📋';
  if (name.endsWith('.sh')) return '💻';
  return '📄';
}

// ==================== FILE MODAL ====================
function openFileModal(file) {
  modalTitle.textContent = file.name;
  modalPath.textContent = file.path || '';

  const ext = file.ext || '';
  if (ext === '.md') {
    modalBody.innerHTML = '<div class="rendered">' + marked.parse(file.content || '') + '</div>';
  } else if (['.yaml', '.yml', '.json', '.py', '.js', '.ts', '.jsx', '.tsx', '.css', '.html', '.sh'].includes(ext)) {
    const langMap = { '.yaml': 'yaml', '.yml': 'yaml', '.json': 'json', '.py': 'python', '.js': 'javascript',
      '.ts': 'typescript', '.jsx': 'javascript', '.tsx': 'typescript', '.css': 'css', '.html': 'html', '.sh': 'bash' };
    const lang = langMap[ext] || '';
    const highlighted = lang && hljs.getLanguage(lang)
      ? hljs.highlight(file.content || '', { language: lang }).value
      : escapeHtml(file.content || '');
    modalBody.innerHTML = `<pre><code class="hljs">${highlighted}</code></pre>`;
  } else {
    modalBody.innerHTML = `<pre><code>${escapeHtml(file.content || '')}</code></pre>`;
  }
  modalOverlay.classList.add('open');
}

function closeModal() {
  modalOverlay.classList.remove('open');
}

// ==================== HASH ROUTING ====================
function handleHash() {
  if (!data) return;
  const hash = window.location.hash.slice(1); // remove #
  if (!hash) return;

  // Find node by path
  for (const top of data.tree) {
    for (const child of (top.children || [])) {
      if (child.path === hash) {
        selectSkill(child, top.name);
        return;
      }
      // Check nested children (for collections with subdirs)
      for (const sub of (child.children || [])) {
        if (sub.path === hash && (sub.type === 'skill' || sub.type === 'agent-doc')) {
          selectSkill(sub, top.name + ' / ' + child.name);
          return;
        }
      }
    }
  }
}
